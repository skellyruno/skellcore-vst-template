cmake_minimum_required(VERSION 3.22)
project(SkellCore VERSION 0.6.0 LANGUAGES CXX)

add_library(SkellCoreDSP INTERFACE)
add_library(SkellCore::DSP ALIAS SkellCoreDSP)
target_compile_features(SkellCoreDSP INTERFACE cxx_std_17)
target_include_directories(SkellCoreDSP INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)

  target_include_directories(SkellCoreUI INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Research snapshots: explicit opt-in includes; keep assertions enabled in Release.
if(SKELLCORE_BUILD_TESTS)
  function(skell_research_test name source group)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE SkellCore::DSP)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/SkellCore/experimental/research/${group})
    if(MSVC)
      target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name} ${ARGN})
  endfunction()
  skell_research_test(SkellResearch_LiveLevelGate tests/research/test_live_level_gate.cpp dynamics)
